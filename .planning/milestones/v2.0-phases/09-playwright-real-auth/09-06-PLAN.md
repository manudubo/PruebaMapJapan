  ```typescript
  import { test, expect } from './fixtures/kc-admin';
  import * as fs from 'fs';
  import * as path from 'path';

  // sessionStorage replay for Playwright bug #31108 — keycloak-js stores tokens here
  const sessionEntries: [string, string][] = (() => {
    try {
      return JSON.parse(
        fs.readFileSync(path.join(__dirname, '../.auth/session.json'), 'utf-8')
      ) as [string, string][];
    } catch {
      return [];
    }
  })();

  test.use({
    storageState: path.join(__dirname, '../.auth/user.json'),
  });

  const FRONTEND_URL = process.env.FRONTEND_URL ?? 'http://localhost:5173';
  const E2E_USERNAME = process.env.E2E_TEST_USERNAME ?? 'e2e-test@local';

  test.describe('Passkey flows', () => {
    // Guard — skip all tests in this describe when KC is not available (D-03)
    // test.skip inside a describe body is valid Playwright API for conditional describe-level skip
    test.skip(!!process.env.SKIP_REAL_AUTH, 'KC not available in this environment');

    test.beforeEach(async ({ context, kcAdmin }) => {
      // Reset credentials before each test — ensures clean state (D-11)
      await kcAdmin.resetCredentials(E2E_USERNAME);

      // Inject sessionStorage tokens before any navigation (D-17: addInitScript before goto)
      if (sessionEntries.length) {
        await context.addInitScript((entries) => {
          for (const [k, v] of entries) {
            window.sessionStorage.setItem(k, v);
          }
        }, sessionEntries);
      }
    });

    test('register passkey via CDP Virtual Authenticator', async ({ page }) => {
      // Navigate to profile page where passkey registration UI lives
      await page.goto(`${FRONTEND_URL}/PruebaMapJapan/profile.html`);
      await page.waitForLoadState('domcontentloaded');
      await page.waitForTimeout(1000);

      // Attach CDP session to the current page
      const cdp = await page.context().newCDPSession(page);
      await cdp.send('WebAuthn.enable', { enableUI: false });
      const { authenticatorId } = await cdp.send('WebAuthn.addVirtualAuthenticator', {
        options: {
          protocol: 'ctap2',
          transport: 'internal',
          hasResidentKey: true,
          hasUserVerification: true,  // CRITICAL: correct spelling (D-10 typo fix)
          isUserVerified: true,
          automaticPresenceSimulation: false,
        },
      });

      // Click the register passkey button in the profile UI
      const registerBtn = page.locator('[data-action="register-passkey"], #register-passkey-btn, button:has-text("Register passkey")');
      await registerBtn.first().click();

      // Wait for KC passkey registration flow to complete and return to profile
      await page.waitForURL(/profile\.html/, { timeout: 30000 });
      await page.waitForLoadState('domcontentloaded');

      // Verify passkey appears in the credentials list
      const credentialsList = page.locator('[data-credential-type="webauthn"], .passkey-item, [data-testid="passkey-credential"]');
      await expect(credentialsList.first()).toBeVisible({ timeout: 10000 });

      // Cleanup virtual authenticator
      await cdp.send('WebAuthn.removeVirtualAuthenticator', { authenticatorId });
    });

    test('login with passkey via KC login form', async ({ page, browser }) => {
      // Step 1: Register a passkey in the authenticated session (beforeEach cleared credentials)
      await page.goto(`${FRONTEND_URL}/PruebaMapJapan/profile.html`);
      await page.waitForLoadState('domcontentloaded');
      await page.waitForTimeout(500);

      const cdpAuth = await page.context().newCDPSession(page);
      await cdpAuth.send('WebAuthn.enable', { enableUI: false });
      const { authenticatorId: authId } = await cdpAuth.send('WebAuthn.addVirtualAuthenticator', {
        options: {
          protocol: 'ctap2',
          transport: 'internal',
          hasResidentKey: true,
          hasUserVerification: true,
          isUserVerified: true,
          automaticPresenceSimulation: false,
        },
      });

      const registerBtn = page.locator('[data-action="register-passkey"], #register-passkey-btn, button:has-text("Register passkey")');
      await registerBtn.first().click();
      await page.waitForURL(/profile\.html/, { timeout: 30000 });

      // Step 2: Capture the registered credential from the authenticated authenticator
      // getCredentials returns the actual CTAP2 credential bytes registered with KC
      const { credentials } = await cdpAuth.send('WebAuthn.getCredentials', { authenticatorId: authId });

      // Step 3: Create a clean browser context — no storageState, no addInitScript
      // addInitScript fires on every navigation in a context, so clearing storage doesn't help.
      // A new context with no configuration is the only way to guarantee KC redirects to login.
      const cleanContext = await browser.newContext();
      const cleanPage = await cleanContext.newPage();

      // Attach CDP to the clean page and inject a fresh virtual authenticator
      const cdpClean = await cleanContext.newCDPSession(cleanPage);
      await cdpClean.send('WebAuthn.enable', { enableUI: false });
      const { authenticatorId: cleanAuthId } = await cdpClean.send('WebAuthn.addVirtualAuthenticator', {
        options: {
          protocol: 'ctap2',
          transport: 'internal',
          hasResidentKey: true,
          hasUserVerification: true,
          isUserVerified: true,
          automaticPresenceSimulation: false,
        },
      });

      // Transfer the registered credential from the auth authenticator to the clean one
      // This makes the clean authenticator respond correctly to KC's WebAuthn assertion challenge
      for (const cred of credentials) {
        await cdpClean.send('WebAuthn.addCredential', {
          authenticatorId: cleanAuthId,
          credential: cred,
        });
      }

      // Step 4: Navigate to dashboard in the clean context — no tokens, so KC redirects to login
      await cleanPage.goto(`${FRONTEND_URL}/PruebaMapJapan/dashboard.html`);
      await cleanPage.waitForURL(/realms\/japan-trip/, { timeout: 15000 });

      // Step 5: On KC login, click passkey/passwordless option if present
      // KC may auto-trigger WebAuthn assertion; the virtual authenticator handles it
      const passkeyBtn = cleanPage.locator(
        '[id*="passkey"], button:has-text("passkey"), button:has-text("Passkey"), a:has-text("Sign in without password")'
      );
      try {
        await passkeyBtn.first().click({ timeout: 3000 });
      } catch {
        // KC initiated WebAuthn assertion automatically via discoverable credential
      }

      // Step 6: After CDP auto-assertion, KC redirects back to app
      await cleanPage.waitForURL(/dashboard\.html/, { timeout: 30000 });
      await cleanPage.waitForLoadState('domcontentloaded');

      // Verify we landed on dashboard (passkey login succeeded)
      expect(cleanPage.url()).toContain('dashboard.html');

      // Cleanup
      await cleanContext.close();
      await cdpAuth.send('WebAuthn.removeVirtualAuthenticator', { authenticatorId: authId });
    });

    test('delete passkey is blocked when it is the last credential', async ({ page }) => {
      // Register one passkey to have something to delete
      await page.goto(`${FRONTEND_URL}/PruebaMapJapan/profile.html`);
      await page.waitForLoadState('domcontentloaded');
      await page.waitForTimeout(500);

      const cdp = await page.context().newCDPSession(page);
      await cdp.send('WebAuthn.enable', { enableUI: false });
      const { authenticatorId } = await cdp.send('WebAuthn.addVirtualAuthenticator', {
        options: {
          protocol: 'ctap2',
          transport: 'internal',
          hasResidentKey: true,
          hasUserVerification: true,
          isUserVerified: true,
          automaticPresenceSimulation: false,
        },
      });

      const registerBtn = page.locator('[data-action="register-passkey"], #register-passkey-btn, button:has-text("Register passkey")');
      await registerBtn.first().click();
      await page.waitForURL(/profile\.html/, { timeout: 30000 });
      await page.waitForLoadState('domcontentloaded');

      // Try to delete the only passkey — last-credential guard (PASS-06) should block this
      const deleteBtn = page.locator('[data-action="delete-passkey"], .delete-passkey-btn, button:has-text("Delete")');
      await deleteBtn.first().click();

      // Last-credential guard must show an error modal or message
      const guardMsg = page.locator(
        '[data-testid="last-credential-error"], .last-credential-warning, .error-message, [role="alert"]'
      );
      await expect(guardMsg.first()).toBeVisible({ timeout: 5000 });

      // The passkey must still be listed (not deleted)
      const credentialsList = page.locator('[data-credential-type="webauthn"], .passkey-item');
      await expect(credentialsList.first()).toBeVisible({ timeout: 5000 });

      // Cleanup
      await cdp.send('WebAuthn.removeVirtualAuthenticator', { authenticatorId });
    });
  });
  ```

  Implementation notes:
  - `test.skip(!!process.env.SKIP_REAL_AUTH, ...)` is inside the describe block — this is valid Playwright API. Playwright evaluates the skip condition at describe-body execution time during test collection.
  - `test.use({ storageState })` applies to all tests in the file. The login-with-passkey test creates a `browser.newContext()` (clean, no storageState) for the unauthenticated KC redirect — `test.use` only applies to the default fixture context, not manually created contexts.
  - `resetCredentials()` is called with `E2E_USERNAME` — the primary test user's passkeys are reset before each test.
  - `addInitScript` fires on every navigation in a context (by design in Playwright). Clearing sessionStorage before `page.goto()` in the same context would be re-populated immediately. The two-context approach (auth context registers, clean context logs in) is the only correct solution.
  - CDP `hasUserVerification` is spelled correctly (NOT the D-10 typo `haUserVerification`).
  - `WebAuthn.getCredentials` returns the raw CTAP2 credential bytes; `WebAuthn.addCredential` injects them into the clean authenticator. This makes the clean authenticator respond correctly to KC's assertion challenge for the specific credential registered in step 1.
  - `automaticPresenceSimulation: false` — KC drives the WebAuthn ceremony; the virtual authenticator responds automatically.
  - Locator strategies use data attributes, IDs, and text — adjust to actual DOM selectors from the profile.ts implementation in Phase 4/8.
