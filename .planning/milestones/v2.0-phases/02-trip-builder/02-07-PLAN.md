6. **"Generar todos los días" — smart merge** (CRITICAL):
   ```typescript
   async function generateDays(dest: ApiDestination, tripId: string): Promise<void> {
     if (!dest.start_date || !dest.end_date) {
       // Show error: destination has no date range
       return;
     }
     // Build set of existing day dates for this destination
     const existingDates = new Set(dest.days.map((d) => d.date));

     // Iterate from start_date to end_date inclusive (both endpoints generate a day)
     const current = new Date(dest.start_date);
     const end = new Date(dest.end_date);
     const promises: Promise<ApiDay>[] = [];

     while (current <= end) {
       const iso = current.toISOString().slice(0, 10); // 'YYYY-MM-DD'
       if (!existingDates.has(iso)) {
         promises.push(createDay(tripId, dest.id, { date: iso }));
       }
       current.setDate(current.getDate() + 1);
     }

     await Promise.all(promises);
     // Re-render days section with updated list
   }
   ```
