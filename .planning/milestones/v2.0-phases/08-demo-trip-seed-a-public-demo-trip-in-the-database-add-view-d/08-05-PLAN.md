```typescript
import { keycloak } from '@/auth/keycloak';

/**
 * Post-login passkey campaign check. (PASS-04)
 *
 * Called from dashboard.ts after initKeycloak() resolves true and getUserInfo()
 * provides a valid userId (D-11).
 *
 * Redirects to the webauthn-register-passwordless AIA if:
 * 1. This device supports WebAuthn (PublicKeyCredential defined)
 * 2. The per-device opt-out cookie has not been set
 *
 * Cookie written BEFORE redirect to prevent a redirect loop if KC
 * registration fails silently or the user presses back (D-14).
 */
export function checkPasskeyCampaign(userId: string): void {
  // D-12: WebAuthn capability check
  if (typeof PublicKeyCredential === 'undefined') return;

  // D-13: per-device cookie check — include equals sign to avoid userId prefix collisions
  if (document.cookie.includes(`pnk_${userId}=`)) return;

  // D-14: write cookie BEFORE redirect
  // No Secure flag — would block cookie on http://localhost
  document.cookie = `pnk_${userId}=1; max-age=2592000; SameSite=Strict`;

  // D-15: redirect to passkey registration AIA
  void keycloak.login({
    action: 'webauthn-register-passwordless',
    redirectUri: window.location.href,
  });
}
```
