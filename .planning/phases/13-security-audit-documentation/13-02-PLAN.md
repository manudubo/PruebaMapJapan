  const signingInput = `${encodedHeader}.${encodedPayload}`;

  const signatureBytes = base64urlToArrayBuffer(encodedSignature);

        { name: 'RSASSA-PKCS1-v1_5' },
        retryKey,
        signatureBytes,
        signingInputBytes,
      );
    }
  }

  if (!isValid) {

```typescript
describe('verifyJwt — JWKS retry on signature failure (SEC-02)', () => {
  const FAKE_KID = 'test-key-id';

  // Minimal valid-structure JWT (header.payload.signature) — real RS256 sig not needed
  // because we mock crypto.subtle.verify to control the outcome
  function makeFakeJwt(kid: string): string {
    const header = btoa(JSON.stringify({ alg: 'RS256', typ: 'JWT', kid }))
      .replace(/\+/g, '-').replace(/\//g, '_').replace(/=/g, '');
    const now = Math.floor(Date.now() / 1000);
    const payload = btoa(JSON.stringify({
      iss: 'http://localhost:8080/realms/japan-trip',
      sub: 'test-user-id',
      aud: 'japan-trip-frontend',
      exp: now + 3600,
      nbf: now - 1,
      iat: now,
    })).replace(/\+/g, '-').replace(/\//g, '_').replace(/=/g, '');
    const sig = 'fakesignature';
    return `${header}.${payload}.${sig}`;
  }

  const mockEnv = {
    KEYCLOAK_URL: 'http://localhost:8080',
    KEYCLOAK_REALM: 'japan-trip',
    VALID_AUDIENCES: 'japan-trip-frontend',
    DATABASE_URL: 'postgresql://mock:mock@localhost/mockdb',
    KC_ADMIN_CLIENT_ID: 'japan-trip-worker',
    KC_ADMIN_CLIENT_SECRET: 'mock-secret',
    OTP_SECRET: 'a3f8c2d1e4b7f0a9d6c3e8b1f4a7d0c2e5b8f3a6d9c0e3b6f1a4d7c0e3b6f1',
  };

  // A fake CryptoKey-like object (only needs to satisfy Map<string, CryptoKey>)
  const fakeCryptoKey = {} as CryptoKey;

  // JWKS JSON that getKeycloakJwks will parse — one RSA key with FAKE_KID
  const fakeJwksJson = {
    keys: [{
      kid: FAKE_KID,
      kty: 'RSA',
      alg: 'RS256',
      use: 'sig',
      n: 'sGb_fake_n_value',
      e: 'AQAB',
    }],
  };

  beforeEach(() => {
    // Reset module-level jwksCache between tests by invalidating via a fresh fetch stub
    vi.stubGlobal('fetch', vi.fn().mockResolvedValue({
      ok: true,
      json: async () => fakeJwksJson,
    }));

    // Mock crypto.subtle.importKey to return our fake CryptoKey
    vi.stubGlobal('crypto', {
      subtle: {
        importKey: vi.fn().mockResolvedValue(fakeCryptoKey),
        verify: vi.fn(),
      },
    });
  });

  afterEach(() => {
    vi.unstubAllGlobals();
  });

  it('retries JWKS fetch and succeeds when signature verification passes on retry', async () => {
    // First verify call returns false (stale key), second returns true (fresh key)
    (crypto.subtle.verify as ReturnType<typeof vi.fn>)
      .mockResolvedValueOnce(false)
      .mockResolvedValueOnce(true);

    const token = makeFakeJwt(FAKE_KID);
    // Should resolve without throwing — retry succeeded
    const payload = await verifyJwt(token, mockEnv as unknown as import('../types').Env);
    expect(payload.sub).toBe('test-user-id');

    // fetch must have been called exactly twice: once for initial JWKS, once for retry
    expect(vi.mocked(fetch)).toHaveBeenCalledTimes(2);
  });

  it('throws after retry if signature still invalid (not a rotation issue)', async () => {
    // Both verify calls return false — token is genuinely invalid
    (crypto.subtle.verify as ReturnType<typeof vi.fn>)
      .mockResolvedValue(false);

    const token = makeFakeJwt(FAKE_KID);
    await expect(verifyJwt(token, mockEnv as unknown as import('../types').Env))
      .rejects.toThrow('JWT signature verification failed');

    // fetch called twice: initial JWKS + one retry (jwksRefreshed guard prevents more)
    expect(vi.mocked(fetch)).toHaveBeenCalledTimes(2);
  });
});
```

Note: The module-level `jwksCache` in keycloak.ts persists between test runs if tests run in the same Worker isolate. The `beforeEach` stub resets `fetch` so that any cached JWKS from a prior test is overwritten on the next `getKeycloakJwks` call triggered by the signature-failure retry. If the cache TTL hasn't expired between tests, the initial `crypto.subtle.verify` call will use the cached key from a prior test — this is acceptable behavior for the test since we control the outcome via the mock.

If tests are flaky due to cache state: add `import { vi } from 'vitest'` and call `vi.resetModules()` in `beforeEach`. This forces a fresh module import and a fresh `jwksCache = null` state.

| T-13-05 | Denial of Service | verifyJwt — JWKS refetch on signature failure | mitigate | `jwksRefreshed` flag limits refetch to exactly 1 per request — attacker sending N forged tokens triggers N Keycloak JWKS fetches max, not N² |
