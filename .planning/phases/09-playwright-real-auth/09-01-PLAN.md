    ```typescript
    const MAILPIT_URL = process.env.MAILPIT_URL ?? 'http://localhost:8025';

    // Replace MailpitMessage and MailpitBody with the actual shape from the spike
    interface MailpitMessage {
      ID: string;  // adjust key name if spike shows different casing
      // other fields (Subject, From, etc.) are optional — only ID is needed
    }

    interface MailpitListResponse {
      messages: MailpitMessage[];  // adjust wrapper key if spike shows different structure
    }

    interface MailpitMessageBody {
      Text: string;  // adjust if spike shows different key (Body.Text, text, etc.)
    }

    export async function purgeInbox(): Promise<void> {
      await fetch(`${MAILPIT_URL}/api/v1/messages`, { method: 'DELETE' });
    }

    export async function fetchLatestOtp(): Promise<string> {
      const res = await fetch(`${MAILPIT_URL}/api/v1/messages`);
      const data = await res.json() as MailpitListResponse;
      if (!data.messages?.length) throw new Error('No messages in Mailpit inbox');
      const msgId = data.messages[0].ID;  // adjust field name per spike
      const msgRes = await fetch(`${MAILPIT_URL}/api/v1/message/${msgId}`);
      const msg = await msgRes.json() as MailpitMessageBody;
      const match = msg.Text.match(/(\d{6})/);  // adjust field access per spike
      if (!match) throw new Error('No 6-digit OTP found in message body');
      return match[1];
    }
    ```
