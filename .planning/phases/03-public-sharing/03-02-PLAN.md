       /**
        * Return a public trip by its UUID slug.
        * Only returns trips with is_public = true.
        */
       export async function getTripBySlug(db: Db, slug: string) {
         return db.query.trips.findFirst({
           where: and(eq(trips.public_slug, slug), eq(trips.is_public, true)),
           with: {
             destinations: {
               orderBy: (d, { asc }) => [asc(d.order_index)],
               with: {
                 hotel: true,
                 days: {
                   orderBy: (d, { asc }) => [asc(d.order_index)],
                   with: {
                     activities: {
                       orderBy: (a, { asc }) => [asc(a.order_index)],
                     },
                   },
                 },
               },
             },
           },
         });
       }

## Trust Boundaries

| Application → PostgreSQL | getTripBySlug passes slug as parameterized query value |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-02-01 | Information Disclosure | GET /api/public/trips/:slug | mitigate | `is_public = true` filter in getTripBySlug WHERE clause — private trips return 404 even if slug is known |
| T-03-02-02 | Spoofing | UUID slug unguessability | accept | `crypto.randomUUID()` produces 122 bits of entropy — brute-force enumeration is computationally infeasible |
| T-03-02-03 | Tampering | CreateTripSchema / UpdateTripSchema | mitigate | `public_slug` absent from both schemas — DB-generated only, clients cannot set or override it |
| T-03-02-04 | Elevation of Privilege | UUID validation bypass | mitigate | `/^[0-9a-f-]{36}$/` regex rejects non-UUID strings before DB query; Drizzle parameterized queries prevent SQL injection |
</threat_model>
